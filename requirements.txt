# =============================================================================
# fiber-fe — dependency manifest
# =============================================================================
#
# NOT a pip requirements file. This is the frontend: plain HTML, CSS and
# JavaScript with no build step and no bundler, so there is nothing to
# `pip install` and nothing to compile. This file exists to state, in one
# place, everything the frontend needs to run — the libraries it vendors, the
# one script it loads from a CDN, the map tiles it draws, the API it talks to
# and the toolchain used for lint/format.
#
# The machine-readable dependency file is `package.json` (with
# `package-lock.json`). It lists the LINT TOOLING ONLY — none of it is loaded
# by the browser at runtime.
#
#   cd fiber-fe && npm ci     # installs the toolchain (Node >= 20)
#
# The site itself needs no install step: copy the directory and serve it.
# =============================================================================


# -----------------------------------------------------------------------------
# 1. Browser runtime
# -----------------------------------------------------------------------------
browser = modern evergreen (Chrome / Edge / Firefox / Safari), ES2020+
build-step = none


# -----------------------------------------------------------------------------
# 2. Vendored libraries (shipped in repo, loaded locally)
# -----------------------------------------------------------------------------
# Source: fiber-fe/vendor/ — byte-for-byte copies of the published dist files,
# including license headers. See vendor/README.md, which records how to update
# them. Self-hosted on purpose: the unpkg CDN was unreliable in some
# environments, and a blocked CDN broke the map on every page.
maplibre-gl-js = 4.7.1        # vendor/maplibre/maplibre-gl.js + maplibre-gl.css
maplibre-license = BSD-3-Clause


# -----------------------------------------------------------------------------
# 3. CDN libraries (loaded over the network at runtime)
# -----------------------------------------------------------------------------
# The only one. Requires outbound HTTPS access to cdnjs from the browser.
exif-js = 2.3.0               # https://cdnjs.cloudflare.com/ajax/libs/exif-js/2.3.0/exif.min.js


# -----------------------------------------------------------------------------
# 4. External map tiles (no API key, but public internet required)
# -----------------------------------------------------------------------------
# Basemap only. Project geometry comes from the platform API, never from here.
carto-basemap = raster tiles, no key
  dark  = https://a.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}@2x.png
  light = https://a.basemaps.cartocdn.com/light_all/{z}/{x}/{y}@2x.png


# -----------------------------------------------------------------------------
# 5. Backend API (must be reachable, or every page renders empty)
# -----------------------------------------------------------------------------
# `fiber-api.js` picks the base URL automatically: served on :8000 it uses the
# same origin; on any other local port it falls back to http://localhost:8000;
# deployed, it uses the production host. The frontend has no base URL of its
# own and needs none — there is one API, and the JWT is issued by it.
fiber-backend = required (Django/DRF, port 8000 locally)
  auth     = JWT bearer, 12h access / 30d refresh
  transport = JSON over HTTPS; multipart/form-data for photo uploads

# The import service (:8180) is called by Django server-side. The browser never
# contacts it, so it is not a frontend requirement.
# The pipeline engine (:8080) and the survey app (:8081) are likewise not
# loaded by these pages.


# -----------------------------------------------------------------------------
# 6. Toolchain (development and CI only — never shipped to the browser)
# -----------------------------------------------------------------------------
node = >= 20                  # package.json "engines"; CI uses node:20-alpine
eslint = ^9.9.0
globals = ^15.9.0             # ESLint globals for browser code
prettier = ^3.3.3
stylelint = ^16.8.0
stylelint-config-standard = ^36.0.0
stylelint-prettier = ^5.0.0

# Commands (see package.json "scripts"):
  npm run lint          # eslint + stylelint + prettier --check
  npm run format        # prettier --write
  npm run ci:check      # what the GitLab lint job runs
  npm run ci:hygiene    # fails if a tracked file matches an ignore rule


# -----------------------------------------------------------------------------
# 7. Serving it locally
# -----------------------------------------------------------------------------
# Three options; none of them installs a package.
#   a) Django serves these pages directly  -> http://localhost:8000
#      (WhiteNoise mounts fiber-fe/, one origin for pages and API)
#   b) node ../server.js from inside fiber-fe/ -> http://localhost:3000
#      (zero-dependency static server using Node's http/fs/path)
#   c) GitLab Pages, which copies the files verbatim (fiber-fe/.gitlab-ci.yml)
